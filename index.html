  • MIPI DSI — Packet Sequence Item + ECC
    SYSTEMVERILOG
    📖 MIPI DSI (Display Serial Interface) uses D-PHY with 1 clock lane + up to 4 data lanes. Packets have DI (VC[7:6]+DT[5:0]), WC (word count), ECC (6-bit Hamming), and CRC. Two modes: High-Speed (HS) and Low-Power (LP/escape).
    typedef enum logic [1:0] {
  VC0=2'b00, VC1=2'b01, VC2=2'b10, VC3=2'b11
} dsi_vc_e;

typedef enum logic [5:0] {
  DT_VSYNC_START=6'h01, DT_HSYNC_START=6'h21,
  DT_RGB888=6'h3E, DT_GEN_LONG_WRITE=6'h29
} dsi_datatype_e;

class dsi_packet extends uvm_sequence_item;
  `uvm_object_utils(dsi_packet)
  rand dsi_vc_e       vc;
  rand dsi_datatype_e dtype;
  rand logic [15:0]   word_count;
  rand byte unsigned  payload[];
  logic [7:0]         ecc;
  logic [15:0]        checksum;
  rand bit            is_hs_mode;

  constraint c_wc_payload { payload.size() == word_count; }
  constraint c_wc_range   { word_count inside {[1:256]}; }
  constraint c_hs_mode    { is_hs_mode dist {1:=80, 0:=20}; }

  // ECC per MIPI DSI D-PHY Annex A
  function logic [7:0] gen_ecc(logic [7:0] di, logic [15:0] wc);
    logic [23:0] d = {wc, di};
    logic [7:0]  e;
    e[0]=d[0]^d[1]^d[2]^d[4]^d[5]^d[7]^d[10]^d[11]^d[13]^d[16]^d[20]^d[21]^d[22]^d[23];
    e[1]=d[0]^d[1]^d[3]^d[4]^d[6]^d[8]^d[10]^d[12]^d[14]^d[17]^d[20]^d[21]^d[22]^d[23];
    e[2]=d[0]^d[2]^d[3]^d[5]^d[6]^d[9]^d[11]^d[12]^d[15]^d[18]^d[20]^d[21]^d[22]^d[23];
    e[3]=d[1]^d[2]^d[3]^d[7]^d[8]^d[9]^d[13]^d[14]^d[15]^d[19]^d[20]^d[21]^d[22]^d[23];
    e[4]=d[4]^d[5]^d[6]^d[7]^d[8]^d[9]^d[16]^d[17]^d[18]^d[19]^d[20]^d[21]^d[22]^d[23];
    e[5]=d[10]^d[11]^d[12]^d[13]^d[14]^d[15]^d[16]^d[17]^d[18]^d[19]^d[20]^d[21]^d[22]^d[23];
    e[6]=0; e[7]=0;
    return e;
  endfunction

  function void post_randomize();
    ecc = gen_ecc({vc,dtype}, word_count);
  endfunction
endclass
  • QSPI — Sequence Item + Driver
    SYSTEMVERILOG
    📖 QSPI extends SPI with 4 bidirectional I/O lines (IO0–IO3) for x1/x2/x4 transfers. Supports Fast-Read, Quad-IO-Read, Page-Program, Sector-Erase commands. Common for NOR Flash (e.g. Winbond W25Q series). Dummy cycles separate command/address from data in high-speed reads.
    typedef enum logic [7:0] {
  QSPI_CMD_PAGE_PROG    = 8'h02,
  QSPI_CMD_FAST_READ    = 8'h0B,
  QSPI_CMD_SECTOR_ERASE = 8'hD8,
  QSPI_CMD_QUAD_IO_READ = 8'hEB
} qspi_cmd_e;

class qspi_seq_item extends uvm_sequence_item;
  `uvm_object_utils(qspi_seq_item)
  rand qspi_cmd_e   cmd;
  rand logic [23:0] addr;
  rand byte         data[];
  rand int unsigned dummy_cycles;
  rand bit          is_read;

  constraint c_page_boundary {
    (cmd == QSPI_CMD_PAGE_PROG) ->
      (addr[7:0] + data.size()) <= 256;
  }
  constraint c_sector_align {
    (cmd == QSPI_CMD_SECTOR_ERASE) -> addr[15:0] == 16'h0;
  }
  constraint c_fast_dummy { (cmd==QSPI_CMD_FAST_READ) -> dummy_cycles==8; }
  constraint c_data_size  { data.size() inside {[1:256]}; }
endclass

  • PLL — Seq Item + Scoreboard + Constraints
    SYSTEMVERILOG
    📖 PLL config exposed via APB: CTRL (EN/BYPASS/SRST/FRAC_EN), NDIV (pre-divider), MDIV (feedback), PDIV (post-divider), FDIV (fractional), STATUS (LOCK/LOST_LOCK/VCO_OOR). Fout = Fref * M / (N * P). VCO must stay in 800–3200 MHz range.
    class pll_seq_item extends uvm_sequence_item;
  `uvm_object_utils(pll_seq_item)
  rand logic [7:0]  ndiv;
  rand logic [11:0] mdiv;
  rand logic [5:0]  pdiv;
  rand logic        frac_en, bypass;
  rand logic [23:0] fdiv;
  rand int          fref_mhz;

  // VCO must be 800–3200 MHz
  constraint c_vco_lo  { (fref_mhz*mdiv/ndiv) >= 800; }
  constraint c_vco_hi  { (fref_mhz*mdiv/ndiv) <= 3200; }
  // Fout = 100–800 MHz
  constraint c_fout    { (fref_mhz*mdiv/(ndiv*pdiv)) inside {[100:800]}; }
  constraint c_ndiv    { ndiv inside {[1:16]}; }
  constraint c_mdiv    { mdiv inside {[16:512]}; }
  constraint c_pdiv    { pdiv inside {[1:8]}; }
  constraint c_fref    { fref_mhz inside {24,25,26,38,40,52}; }

  function real get_fout_mhz();
    return real'(fref_mhz)*real'(mdiv) /
           (real'(ndiv)*real'(pdiv));
  endfunction
endclass

class pll_scoreboard extends uvm_scoreboard;
  function void write(pll_seq_item item);
    real vco = real'(item.fref_mhz)*real'(item.mdiv)/real'(item.ndiv);
    if (vco<800.0 || vco>3200.0)
      `uvm_error("PLL_SB",$sformatf("VCO OOR: %.1f MHz",vco))
    else
      `uvm_info("PLL_SB",$sformatf("OK fout=%.2f MHz",item.get_fout_mhz()),UVM_MEDIUM)
  endfunction
endclass
  • AHB — Burst Sequence + Driver
    SYSTEMVERILOG
    📖 AHB-Lite: transfers use HTRANS (IDLE/BUSY/NONSEQ/SEQ), HBURST (SINGLE/INCR/WRAP4/8/16), HSIZE (byte/hword/word). Wrapping bursts require naturally-aligned start address. HREADY extends transfer; HRESP signals error.
    typedef enum logic [1:0] {
  AHB_IDLE=2'b00, AHB_NONSEQ=2'b10, AHB_SEQ=2'b11
} ahb_htrans_e;

typedef enum logic [2:0] {
  AHB_SINGLE=3'b000, AHB_WRAP4=3'b010,
  AHB_INCR4=3'b011,  AHB_WRAP8=3'b100,
  AHB_INCR8=3'b101,  AHB_WRAP16=3'b110
} ahb_hburst_e;

class ahb_seq_item extends uvm_sequence_item;
  `uvm_object_utils(ahb_seq_item)
  rand logic [31:0]  haddr;
  rand logic [31:0]  hwdata;
  rand logic         hwrite;
  rand ahb_hburst_e  hburst;
  rand logic [2:0]   hsize;

  // Natural alignment for wrapping bursts
  constraint c_wrap4  { (hburst==AHB_WRAP4)  -> haddr[3:0]==4'h0; }
  constraint c_wrap8  { (hburst==AHB_WRAP8)  -> haddr[4:0]==5'h00; }
  constraint c_wrap16 { (hburst==AHB_WRAP16) -> haddr[5:0]==6'h00; }
  constraint c_size   {
    (hsize==3'b001) -> haddr[0]==1'b0;    // halfword
    (hsize==3'b010) -> haddr[1:0]==2'b00; // word
  }
endclass

// AHB burst sequence with wrap address calculation
class ahb_burst_seq extends uvm_sequence #(ahb_seq_item);
  `uvm_object_utils(ahb_burst_seq)
  rand ahb_hburst_e burst_type;
  rand logic [31:0] start_addr;

  task body();
    ahb_seq_item item;
    int beats, step;
    logic [31:0] addr = start_addr;
    beats = (burst_type==AHB_WRAP4||burst_type==AHB_INCR4)  ? 4  :
            (burst_type==AHB_WRAP8||burst_type==AHB_INCR8)  ? 8  :
            (burst_type==AHB_WRAP16)                         ? 16 : 1;
    step = 4; // word
    for (int i=0; i<beats; i++) begin
      item = ahb_seq_item::type_id::create($sformatf("b%0d",i));
      start_item(item);
      void'(item.randomize() with {haddr==addr; hburst==burst_type;});
      finish_item(item);
      if (burst_type inside {AHB_WRAP4,AHB_WRAP8,AHB_WRAP16}) begin
        int wsz = beats*step;
        addr = (addr & ~(wsz-1)) | ((addr+step) & (wsz-1));
      end else addr += step;
    end
  endtask
endclass

DPI allows the user to easily call functions of other launguage from SystemVerilog and to export SystemVerilog functions, so that they can be called in other launguages.

DPI Import and export methods

Import Method
The methods (functions/tasks) implemented in Foreign language can be called from SystemVerilog and such methods are called Import methods.

Export methods
The methods implemented in SystemVerilog can be called from Foreign language such methods are called Export methods.

  • shallow_vs_deep_copy.sv

    Shallow copy (default new) copies handles not objects — nested object changes affect both copies. Deep copy recursively clones nested objects.

    SV CONCEPTS 
    class inner_c;
  int val;
endclass

class outer_c;
  int       data;
  inner_c   inner;  // nested object

  // SHALLOW COPY — built-in
  // outer_c copy = new src;
  // copy.inner IS SAME OBJECT as src.inner!

  // DEEP COPY — manual
  function outer_c deep_copy();
    outer_c c  = new this;     // copy primitives
    c.inner    = new inner;    // clone nested object
    return c;
  endfunction
endclass

outer_c src  = new();
src.inner    = new();
src.inner.val = 5;

outer_c sh   = new src;         // shallow
outer_c dp   = src.deep_copy(); // deep

sh.inner.val = 99;  // also changes src.inner.val!
dp.inner.val = 99;  // does NOT change src.inner.val

  • Shallow Copy vs Deep Copy
    SYSTEMVERILOG
    📖 Shallow copy (default new) copies handles, not objects — nested objects are shared. Deep copy requires a custom copy() method that creates new instances of nested objects.
    class inner;
  int data;
endclass

class outer;
  inner obj;
  int   id;

  function outer shallow_copy();
    outer tmp = new this;   // built-in shallow copy
    return tmp;
  endfunction

    outer tmp   = new();

    return tmp;
  endfunction
