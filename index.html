    SYSTEMVERILOG 
    // AHB Wrap Burst Transaction Class
// Covers WRAP4, WRAP8, WRAP16 burst types

class ahb_wrap_burst;

  rand bit [31:0] start_addr;
  rand bit [2:0]  hsize;    // 0=byte, 1=half, 2=word
  rand bit [2:0]  hburst;   // WRAP4/8/16
  rand int        burst_len;

  // Constrain to wrap burst types only
  constraint c_burst_type {
    hburst inside {3'b010, 3'b100, 3'b110}; // WRAP4/8/16
  }

  constraint c_len {
    burst_len inside {4, 8, 16};
    hburst == 3'b010 -> burst_len == 4;
    hburst == 3'b100 -> burst_len == 8;
    hburst == 3'b110 -> burst_len == 16;
  }

  // Address must be aligned to transfer size
  constraint c_align {
    (start_addr & ((1 << hsize) - 1)) == 0;
  }

  function void display();
    $display("[AHB_WRAP] addr=0x%0h size=%0d len=%0d burst=0x%0b",
             start_addr, hsize, burst_len, hburst);
  endfunction

endclass
  • ahb_base_seq.sv
    SYSTEMVERILOG / UVM 
    // AHB UVM Base Sequence
// Drives INCR and WRAP bursts via sequencer

class ahb_base_seq extends uvm_sequence #(ahb_seq_item);
  `uvm_object_utils(ahb_base_seq)

  function new(string name = "ahb_base_seq");
    super.new(name);
  endfunction

  task body();
    ahb_seq_item req;

    repeat(10) begin
      req = ahb_seq_item::type_id::create("req");
      start_item(req);
      if(!req.randomize() with {
        hburst inside {WRAP4, INCR4, INCR8};
        hsize  == 2'b10; // Word transfers
      })
        `uvm_error(get_name(), "Randomization failed")
      finish_item(req);
    end
  endtask

endclass

    // Constraint: Array Pattern 2-4-6-8-8-6-4-2
// First half: even sequence ascending
// Second half: mirror of first half

class array_pattern;

  rand byte a[20];

  constraint c_mirror_pattern {
    foreach(a[i]) {
      if(i < (a.size() / 2))
        a[i] == (i + 1) * 2;        // 2,4,6,8,10...
      else
        a[i] == a[a.size() - 1 - i]; // mirror
    }
  }

  function void print_array();
    $write("Pattern: ");
    foreach(a[i]) $write("%0d ", a[i]);
    $display();
  endfunction

endclass

    // Weighted Distribution Constraint
// Useful for biasing stimulus toward corner cases

class ahb_data_gen;

  rand bit [31:0] data;
  rand bit [7:0]  addr_offset;

  // 60% normal, 30% max, 10% zero
  constraint c_data_dist {
    data dist {
      [32'h0000_0001 : 32'hFFFE_FFFE] := 60,
      32'hFFFF_FFFF                   := 30,
      32'h0000_0000                   := 10
    };
  }

  // Address aligned to 4-byte boundary
  constraint c_addr_align {
    addr_offset[1:0] == 2'b00;
    addr_offset inside {[8'h00 : 8'hFC]};
  }

endclass
  • unique_no_repeat.sv
    SYSTEMVERILOG 
    // Unique Constraint — No Repeated Addresses
// Ensures all generated addresses are distinct

class unique_addr_gen;

  rand bit [7:0] addr[8];
  rand bit [2:0] priority_level;

  // All 8 addresses must be unique
  constraint c_unique_addr {
    unique { addr };
  }

  // Addresses in ascending order
  constraint c_ascending {
    foreach(addr[i])
      if(i > 0) addr[i] > addr[i-1];
  }

  // Resolve priority before address
  constraint c_solve_order {
    solve priority_level before addr;
  }

endclass

    UVM / SYSTEMVERILOG 
    // Generic UVM Scoreboard
// Compares expected vs actual transactions

class ahb_scoreboard extends uvm_scoreboard;
  `uvm_component_utils(ahb_scoreboard)

  uvm_analysis_imp #(ahb_seq_item, ahb_scoreboard)
    exp_port, act_port;

  ahb_seq_item exp_queue[$], act_queue[$];
  int pass_cnt, fail_cnt;

  function new(string name, uvm_component parent);
    super.new(name, parent);
  endfunction

  function void write_exp(ahb_seq_item t);
    exp_queue.push_back(t);
  endfunction

  function void write_act(ahb_seq_item t);
    act_queue.push_back(t);
    check_transaction();
  endfunction

  function void check_transaction();
    ahb_seq_item exp, act;
    if(exp_queue.size() == 0) return;
    exp = exp_queue.pop_front();
    act = act_queue.pop_front();
    if(exp.data === act.data) begin
      pass_cnt++;
      `uvm_info(get_name(), "PASS: data match", UVM_LOW)
    end else begin
      fail_cnt++;
      `uvm_error(get_name(), $sformatf(
        "FAIL: exp=0x%0h act=0x%0h", exp.data, act.data))
    end
  endfunction

endclass
  • coverage_collector.sv
    SYSTEMVERILOG 
    // Functional Coverage Collector
// Tracks AHB burst types and transfer sizes

class ahb_coverage extends uvm_subscriber #(ahb_seq_item);
  `uvm_component_utils(ahb_coverage)

  ahb_seq_item trans;

  covergroup ahb_cg;

    cp_burst: coverpoint trans.hburst {
      bins single  = {3'b000};
      bins incr    = {3'b001};
      bins wrap4   = {3'b010};
      bins incr4   = {3'b011};
      bins wrap8   = {3'b100};
      bins incr8   = {3'b101};
      bins wrap16  = {3'b110};
      bins incr16  = {3'b111};
    }

    cp_size: coverpoint trans.hsize {
      bins byte_sz = {3'b000};
      bins half_sz = {3'b001};
      bins word_sz = {3'b010};
    }

    cx_burst_size: cross cp_burst, cp_size;

  endgroup

  function new(string name, uvm_component parent);
    super.new(name, parent);
    ahb_cg = new();
  endfunction

  function void write(ahb_seq_item t);
    trans = t;
    ahb_cg.sample();
  endfunction

endclass
