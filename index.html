  • Events — Blocking (->) vs Non-blocking (->>) Trigger
    SYSTEMVERILOG
    📖 Blocking trigger (->): @(e) must be waiting BEFORE ->e fires or the event is missed. Non-blocking (->>): executes in Inactive region, so trigger BEFORE wait is safe — event persists until sampled.
    // Blocking trigger — wait must come FIRST
module ev_blocking;
  event e;
  initial begin
    \$display("Before Event Wait");
    @(e);                       // blocks here
    \$display("After Event Wait");
  end
  initial begin
    \$display("Before Event Trigger");
    -> e;                       // unblocks waiting thread
    \$display("After Event Trigger");
  end
endmodule
// output: Before Event Wait
//         Before Event Trigger
//         After Event Trigger
//         After Event Wait

// Non-blocking trigger — trigger BEFORE wait is OK
module ev_nonblocking;
  event e;
  initial begin
    \$display("Before Event Trigger");
    ->> e;                      // non-blocking, safe
    \$display("After Event Trigger");
    \$display("Before Event Wait");
    @(e);                       // catches it
    \$display("After Event Wait");
  end
endmodule
// output: Before Event Trigger
//         After Event Trigger
//         Before Event Wait
//         After Event Wait

    📖 Functions return a value (void = none), execute in zero time, cannot have delays. Pass by value = copy (changes not visible outside). Pass by ref = shared handle (changes visible). Default args allow calling with fewer arguments.

  x = x + y; // local only
  y = x + y;
  return x + y; // 130
endfunction

// Pass by REFERENCE — changes visible outside
function int sum_ref(ref int x, y);

  y = x + y; // modifies caller's y
  return x + y;
endfunction

// Static vs Automatic variables
function static int incr_static();
  static int count_A;   // PERSISTS between calls
  automatic int count_B; // reset to 0 each call
  int count_C;           // static by default in static func
  count_A++; count_B++; count_C++;
  \$display("Static: A=%0d B=%0d C=%0d", count_A, count_B, count_C);
endfunction
// Call 1: A=1 B=1 C=1
// Call 2: A=2 B=1 C=2  ← B reset, A/C persist

function automatic int incr_auto();
  static int count_A;    // static inside automatic
  automatic int count_B; // reset each call
  count_A++; count_B++;
  \$display("Auto: A=%0d B=%0d", count_A, count_B);
endfunction

  • Fork-Join — Mimicking Variants & disable fork
    SYSTEMVERILOG
    📖 join=wait ALL, join_any=wait FIRST, join_none=fire-and-forget. Mimic join_none with join_any (add empty process). Mimic join_any with join_none (use event). Mimic join with join_any + wait fork.
    // Mimic join_none using join_any
module any_to_none;
  task process_a(); \$display("A @%0t",$time); #5; \$display("A done"); endtask
  task process_b(); \$display("B @%0t",$time); #10; \$display("B done"); endtask
  initial begin
    fork
      begin end      // empty process — exits immediately
      process_a();
      process_b();
    join_any         // exits at t=0 (empty process wins)
    \$display("Out of fork @%0t", \$time); // t=0
  end
endmodule

// Mimic join_any using join_none + event
module none_to_any;
  event e;
  task process_a(); #5; \$display("A done @%0t",$time); ->e; endtask
  task process_b(); #10; \$display("B done @%0t",$time); ->e; endtask
  initial begin
    fork
      process_a();
      process_b();
    join_none
    @e;
    \$display("Out of fork @%0t", \$time); // t=5
  end
endmodule

// Mimic join (wait ALL) using join_any + wait fork
module any_to_all;
  initial begin
    fork
      process_a();
      process_b();
    join_any
    wait fork;       // wait for all remaining threads
    \$display("All done @%0t", \$time); // t=10
  end
endmodule

  • Semaphore — Shared Resource with/without Protection
    SYSTEMVERILOG
    📖 Without semaphore: concurrent processes race — result depends on last writer. With semaphore: get(N) acquires keys (blocks if unavailable), put(N) releases. Serialises access.
    // WITHOUT semaphore — race condition
module without_sem;
  int y;
  task automatic shared_resource(string s, int a, b);
    \$display("%0s: writing started", s);
    #5ns; y = a + b;
    \$display("%0s: at %0t: y=%0d", s, \$time, y);
  endtask
  initial begin
    fork
      shared_resource("1st", 10, 20);
      shared_resource("2nd", 20, 30);
      shared_resource("3rd", 30, 40);
    join
  end
endmodule
// All start at t=0, ALL finish at t=5 with y=70 (last write wins)

// WITH semaphore — serialised access
module with_sem;
  semaphore sem;
  int y;
  task automatic shared_resource(string s, int a, b);
    sem.get(1);
    \$display("%0s: writing started @%0t", s, \$time);
    #5ns; y = a + b;
    \$display("%0s: done @%0t y=%0d", s, \$time, y);
    sem.put();
  endtask
  initial begin
    sem = new(1);
    fork
      shared_resource("1st", 10, 20);
      shared_resource("2nd", 20, 30);
      shared_resource("3rd", 30, 40);
    join
  end
endmodule
// 1st: t=0→5 y=30, 2nd: t=5→10 y=50, 3rd: t=10→15 y=70

  • Functional Coverage — Implicit & Explicit Bins
    SYSTEMVERILOG
    📖 Implicit bins: auto-created up to 64 (auto_bin_max). Variables >6 bits get 64 bins covering ranges. Explicit bins: user-defined ranges, sets, and groups for precise coverage control.

  • Covergroup — Clocking Event, Expressions & Cross Coverage
    SYSTEMVERILOG
    📖 Covergroup with @(posedge clk) auto-samples on the clock edge. Expressions in coverpoints compute values at sample time. Cross coverage tracks all combinations of two or more coverpoints.
    module tb;
  bit [1:0] mode;
  bit [2:0] cfg;
  bit clk;
  always #20 clk = ~clk;

  covergroup cg @(posedge clk);   // auto-sample on clock
    option.per_instance = 1;

    cp_mode : coverpoint mode;             // 4 bins: {0,1,2,3}
    cp_cfg_10 : coverpoint cfg[1:0];      // 4 bins: lower 2 bits
    cp_cfg_lsb : coverpoint cfg[0];       // 2 bins: LSB only
    cp_sum : coverpoint (mode + cfg);     // expression coverpoint
    // (mode 0..3) + (cfg 0..7) = sum 0..10

    // Cross coverage — all combinations of mode X cfg[1:0]
    cx_mode_cfg : cross cp_mode, cp_cfg_10;
    // 4 * 4 = 16 cross bins
  endgroup

  cg cg_inst;
  initial begin
    cg_inst = new();
    for (int i = 0; i < 5; i++) begin
      @(negedge clk);          // sample at negedge, update values
      mode = \$random;
      cfg  = \$random;
      \$display("[%0t] mode=%0h cfg=%0h", \$time, mode, cfg);
    end
  end

  initial #500 \$finish;
endmodule
  • Transition Bins — Consecutive, Goto & Non-Consecutive
    SYSTEMVERILOG
    📖 Transition bins track value sequences over time. [*N] = consecutive N times. [->N:M] = goto: N to M times, immediately followed. [=N:M] = non-consecutive: N to M times, sometime later.
    module coverage;
  bit [2:0] data;

  covergroup c_g;
    option.per_instance = 1;
    coverpoint data {
      // Consecutive repetition: three 5s in a row
      bins b1 = (5[*3]);          // 5 => 5 => 5

      // Goto repetition: 1, then 4-6 non-consecutive 3s immediately followed by 1
      bins b2 = (1 => 3[->4:6] => 1);

      // Non-consecutive: 1, then 3-6 non-consecutive 2s, sometime later 5
      bins b3 = (1 => 2[=3:6] => 5);

      // Goto: 3-5 non-consecutive 2s (goto)
      bins b4 = (2[->3:5]);

      // Goto: 3-5 non-consecutive 3s
      bins b5 = (3[->3:5]);
    }
  endgroup

  initial begin
    c_g cg = new();
    // Stimulus: 1 4 3 2 3 3 2 2 3 2 3 1 5 5 5 5 5 5
    int stim[] = '{1,4,3,2,3,3,2,2,3,2,3,1,5,5,5,5,5,5};
    foreach (stim[i]) begin
      #1 data = stim[i];
      cg.sample();
    end
    #1 \$finish;
  end
endmodule
// b1 covered: 5,5,5 sequence appears
// b2 covered: 1 ... 3(x5) ... 1 sequence
// b3 covered: 1 ... 2(x4) ... 5 sequence
  • ignore_bins, illegal_bins & wildcard bins
    SYSTEMVERILOG
    📖 ignore_bins: excluded from coverage metric (won't count). illegal_bins: trigger a simulation error when hit. wildcard bins: use x/? to match any bit value — one bin covers multiple values.
    module func_coverage;
  logic [3:0] addr;

  covergroup c_group;
    option.per_instance = 1;

    cp1: coverpoint addr {
      // Excluded from coverage — won't affect %
      ignore_bins  exclude  = {8, 10};

      // Error if these values ever occur
      illegal_bins invalid  = {13, 14};

      // Normal range bins
      bins valid_1[4] = {[0:3]};   // 4 values, 4 bins

      // Wildcard bin: 3'b1x? matches {100,101,110,111} = {4,5,6,7}
      wildcard bins wildcard_valid = {3'b1x?};
    }
  endgroup

  c_group cg = new();
  initial
    repeat(20) begin
      #1 addr = \$random;
      cg.sample();
      \$display("addr=%0d", addr);
    end
endmodule
// When addr==13 or 14: simulation ERROR is reported immediately
// When addr==8 or 10: no error, but NOT counted in coverage
// When addr==4,5,6,7: covered by wildcard bin
